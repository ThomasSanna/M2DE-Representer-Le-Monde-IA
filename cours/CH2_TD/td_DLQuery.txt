Exprimez les requêtes précédentes en SPARQL
a) Lister tous les noms des doctorants
PREFIX : <http://www.univ-corse.fr/ontologies/ontocampus#>

SELECT ?x WHERE {
    ?x a :Doctorant
}

b) Lister les titulaires affiliées à l'université de corse
SELECT ?x WHERE {
    ?x a :Titulaire .
    ?x :affilieA :univ-corse
}

c) Lister les doctorants qui participent à un projet
SELECT ?x WHERE {
    ?x a :Doctorant .
    ?x :travailleSur ?y .
    ?y a :Projet
}

d) Lister les titulaires qui sont aussi encadrants
SELECT ?x WHERE {
    ?x a :Encadrant .
    ?x a :Titulaire
}

e) Les projets dirigés par bob
SELECT ?x WHERE {
    ?x a :Projet .
    :bob :dirige ?x
}

f) Les doctorants encadrés par bob
SELECT ?x WHERE {
    ?x a :Doctorant .
    :bob :encadre ?x
}

g) Les doctorants qui n'ont pas encore soutenu
SELECT ?doctorant
WHERE {
  ?doctorant a :Doctorant .
  ?doctorant :statutThese ?statut .
  FILTER (?statut = "en cours")
}

h) Les titulaires qui n'encadrent pas lucas
SELECT ?x
WHERE {
  ?x a :Titulaire .
  ?x :encadre ?y .
  ?y a :Doctorant .
  FILTER NOT EXISTS {
    ?x :encadre :lucas
  }
}
